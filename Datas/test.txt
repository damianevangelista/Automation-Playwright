import { Page, expect, Locator } from "@playwright/test";
import * as allure from "allure-js-commons"

class adminJobs {
    private page: Page;
    private job: Locator;
    private jobTitle: Locator;
    private payGrade: Locator;
    private employmentStatus: Locator;
    private jobCategory: Locator;
    private workShift: Locator;
    private addJobBtn: Locator;
    private addJobTitle: Locator;
    private jobTitleInput: Locator;
    private jobDescriptionInput: Locator;
    private jobFile: Locator;
    private jobNote: Locator;
    private saveJobBtn: Locator;
    private cancelJobBtn: Locator;
    private jobTitleRequiered: Locator;
    private notChosenFile: Locator;


    constructor(page: Page) {
        this.page = page;
        this.job = this.page.locator('li:has-text("Job")');
        this.jobTitle = this.page.getByRole('listitem').filter({ hasText: /^Job Titles$/ });
        this.payGrade = this.page.getByRole('menuitem', { name: 'Pay Grades' });
        this.employmentStatus = this.page.getByRole('menuitem', { name: 'Employment Status' });
        this.jobCategory = this.page.getByRole('menuitem', { name: 'Job Categories' });
        this.workShift = this.page.getByRole('menuitem', { name: 'Work Shifts' });
        this.addJobBtn = this.page.getByRole('button', { name: 'Add' });
        this.addJobTitle = this.page.getByRole('heading', { name: 'Add Job Title' });
        this.jobTitleInput = this.page.getByRole('textbox').nth(1);
        this.jobDescriptionInput = this.page.getByRole('textbox', { name: 'Type description here' });
        this.jobFile = this.page.getByText('Browse');
        this.jobNote = this.page.getByRole('textbox', { name: 'Add note' });
        this.saveJobBtn = this.page.getByRole('button', { name: 'Save' });
        this.cancelJobBtn = this.page.getByRole('button', { name: 'Cancel' });
        this.jobTitleRequiered = this.page.getByText('Required', { exact: true });
        this.notChosenFile = this.page.getByText('No file chosen')
    }

    async addJob(jobtitle: string, jobdescription: string, jobfile: string, jobnote: string) {
        await allure.step('Click on Job', async () => {
            await this.job.click()
        })
        await allure.step('Click on Job Title', async () => {
            await this.jobTitle.click()
        })
        await allure.step('Click on Add Job Title Button', async () => {
            await this.addJobBtn.click()
        })
        await allure.step('Job Title check', async () => {
            await expect(this.addJobTitle).toBeVisible()
        })
        await allure.step('Enter Job Title', async () => {
            await this.jobTitleInput.fill(jobtitle)
        })
        await allure.step('Enter Job Description', async () => {
            await this.jobDescriptionInput.fill(jobdescription)
        })
        await allure.step('Upload Job File', async () => {
            await this.jobFile.setInputFiles(jobfile)
        })
        await allure.step('Job File check', async () => {
            await expect(this.notChosenFile).not.toBeVisible()
        })
        await allure.step('Add Job Note', async () => {
            await this.jobNote.fill(jobnote)
        })
        await allure.step('Click on Save Button', async () => {
            await this.saveJobBtn.click()
        })
    }
}

export default adminJobs;
